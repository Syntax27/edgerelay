cmake_minimum_required(VERSION 3.20)
project(EdgeRelay VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

function(edgerelay_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()

add_executable(edgerelay src/main.cpp src/net.cpp)
target_include_directories(edgerelay PRIVATE include)
target_link_libraries(edgerelay PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(edgerelay PRIVATE ws2_32)
endif()
edgerelay_warnings(edgerelay)

include(CTest)
if(BUILD_TESTING)
    add_executable(policy_tests tests/policy_tests.cpp)
    target_include_directories(policy_tests PRIVATE include)
    target_link_libraries(policy_tests PRIVATE Threads::Threads)
    edgerelay_warnings(policy_tests)
    add_test(NAME policy_tests COMMAND policy_tests)
endif()
